  import random
  from jobqueue import Job, plan_jobs
  rng=random.Random(42)
  for trial in range(300):
      n=rng.randrange(0,15)
      ids=[f"j{i}" for i in range(n)]
      # DAG: deps only on earlier ids, then shuffle input order
      spec={}
      for i,name in enumerate(ids):
          deps=tuple(rng.sample(ids[:i], rng.randrange(0,i+1))) if i else ()
          spec[name]=(deps, rng.choice([-2,0,1,1,5]))
      order=ids[:]; rng.shuffle(order)
      jobs=[Job(name, priority=spec[name][1], depends_on=spec[name][0]) for name in order]
      got=plan_jobs(jobs)
      # independent naive reference
      done=[]; rem=list(jobs)
      while rem:
          ready=[j for j in rem if all(d in done for d in j.depends_on)]
          assert ready
          ready.sort(key=lambda j:-j.priority)  # stable: retains input order among ties
          nxt=ready[0]; done.append(nxt.id); rem.remove(nxt)
      assert got==done, (trial, got, done)
      pos={x:i for i,x in enumerate(got)}
      for j in jobs:
          assert all(pos[d]<pos[j.id] for d in j.depends_on)
  print("randomized DAG check: 300 trials ok")
  PY
